// REPLACE THIS:
function getAllAvailableSections() {
    const savedOrder = localStorage.getItem('sectionOrder');
    if (savedOrder) {
        const sections = JSON.parse(savedOrder) || [];
        return sections.filter(s => s && s.visible !== false).map(s => s.id);
    }
    return [];
}

// WITH THIS:
async function getAllAvailableSections() {
    try {
        if (window.hubDatabase && window.hubDatabaseReady && typeof hubDatabase.getAllSections === 'function') {
            const sections = await hubDatabase.getAllSections();
            return sections.filter(s => s && s.visible !== false).map(s => s.section_id || s.id);
        }
    } catch (error) {
        console.error('Error loading sections from database:', error);
    }
    return [];
}
